32static inline bool addr_is_base58(const char *s)
33{
34 static const char *const B58 =
35 "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz";
36 if (s == NULL) { return false; }
37 /* An empty string has no character outside the alphabet, which would make a
38 * plain loop answer "yes". Callers check the length too, but this is the
39 * function whose name promises the answer, so it answers correctly. */
40 if (*s == '\0') { return false; }
41 for (const char *p = s; *p != '\0'; p++) {
42 if (strchr(B58, *p) == NULL) { return false; }
43 }
44 return true;
45}

66static inline bool addr_tron_decode(const char *s, unsigned char out[25])
67{
68 static const char *const B58 =
69 "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz";
70 if (!addr_tron_plausible(s)) { return false; }
71 memset(out, 0, 25U);
72 for (const char *p = s; *p != '\0'; p++) {
73 unsigned carry = static_cast<unsigned>(strchr(B58, *p) - B58);
74 for (int j = 24; j >= 0; j--) {
75 carry += 58U * out[j];
76 out[j] = static_cast<unsigned char>(carry & 0xFFU);
77 carry >>= 8;
78 }
79 if (carry != 0U) { return false; } /* more than 25 bytes */
80 }
81 return out[0] == 0x41U;
82}
