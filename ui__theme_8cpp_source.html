14/******************************************************************
15 * 3a. LVGL theme — extends the built-in default rather than replacing it
16 *
17 * Restyles the widgets not styled by hand (tab bar as a segmented control,
18 * thin scrollbars, sliders, fields, keys) to match the hand-drawn pills. A
19 * theme applies at creation, so every future tabview gets it too.
20 *
21 * Deliberately narrow: buttons, pills and cards are styled explicitly at each
22 * call site, and a theme fighting an explicit style depends on which ran last.
23 ******************************************************************/

111static void theme_apply(lv_theme_t *th, lv_obj_t *obj)
112{
113 LV_UNUSED(th);
114
115 lv_obj_add_style(obj, &s_st_scrollbar, LV_PART_SCROLLBAR);
116
117 if (lv_obj_check_type(obj, &lv_tabview_class)) {
118 lv_obj_add_style(obj, &s_st_tabview, LV_PART_MAIN);
119 return;
120 }
121
122 /* The tab bar is a button matrix, and so is the PIN pad; the parent tells
123 * them apart, so the keypad does not become a segmented control. */
124 if (lv_obj_check_type(obj, &lv_btnmatrix_class)) {
125 lv_obj_t *parent = lv_obj_get_parent(obj);
126 if ((parent != NULL) && lv_obj_check_type(parent, &lv_tabview_class)) {
127 lv_obj_add_style(obj, &s_st_tabbar, LV_PART_MAIN);
128 lv_obj_add_style(obj, &s_st_tab, LV_PART_ITEMS);
129 lv_obj_add_style(obj, &s_st_tab_sel,
130 LV_PART_ITEMS | LV_STATE_CHECKED);
131 }
132 return;
133 }
134
135 if (lv_obj_check_type(obj, &lv_slider_class)) {
136 lv_obj_add_style(obj, &s_st_track, LV_PART_MAIN);
137 lv_obj_add_style(obj, &s_st_ink, LV_PART_INDICATOR);
138 lv_obj_add_style(obj, &s_st_ink, LV_PART_KNOB);
139 } else if (lv_obj_check_type(obj, &lv_arc_class)
140 || lv_obj_check_type(obj, &lv_spinner_class)) {
141 lv_obj_add_style(obj, &s_st_track, LV_PART_MAIN);
142 lv_obj_add_style(obj, &s_st_ink, LV_PART_INDICATOR);
143 } else if (lv_obj_check_type(obj, &lv_textarea_class)) {
144 lv_obj_add_style(obj, &s_st_field, LV_PART_MAIN);
145 } else if (lv_obj_check_type(obj, &lv_keyboard_class)) {
146 lv_obj_add_style(obj, &s_st_field, LV_PART_MAIN);
147 lv_obj_add_style(obj, &s_st_key, LV_PART_ITEMS);
148 }
149}

157void theme_init(void)
158{
159 theme_styles_init();
160
161 /* Re-initialised with the UI font: the one LVGL built it with is a
162 * Kconfig built-in, and every widget not styled by hand inherits it. */
163 lv_theme_t *base = lv_disp_get_theme(NULL);
164 base = lv_theme_default_init(NULL, base->color_primary, base->color_secondary,
165 LV_THEME_DEFAULT_DARK, &font_inter_14);
166 s_theme = *base;
167 lv_theme_set_parent(&s_theme, base);
168 lv_theme_set_apply_cb(&s_theme, theme_apply);
169 lv_disp_set_theme(NULL, &s_theme);
170}
