216/* A payout address or a token contract proposed from a browser, shown for
217 * acceptance here. The modal is the security boundary, not decoration: a browser
218 * can propose a value, only the panel in the operator's hands can store one — and
219 * that holds for the card-derived route too, which comes through the same
220 * handshake rather than writing straight to NVS.
221 *
222 * The value is drawn wrapped at 14 px rather than elided — the operator is being
223 * asked to compare it against their own records character by character, so every
224 * character has to be on the screen. */
225void build_prov_confirm(void) {
226 prov_ask_t kind = PROV_ASK_NONE;
227 char label[40] = "";
228 char addr[SETTINGS_PAYOUT_MAX] = "";
229 if (!prov_pending(&kind, label, sizeof(label), addr, sizeof(addr))) { return; }
230
231 const bool contract = (kind == PROV_ASK_CONTRACT_ETH) ||
232 (kind == PROV_ASK_CONTRACT_TRON);
233
234 /* Stacked and measured with ota_text/ota_fit_card, because the label, the
235 * address and the warning all vary in height, and a fixed layout lets a
236 * taller block run under the Accept button on the screen that decides where
237 * the takings go. The label wraps rather than eliding, for the same reason
238 * the address does. */
239 lv_obj_t *card = open_modal(OTA_CARD_W, 276);
240
241 lv_obj_t *t = ota_text(card, NULL,
242 contract ? "Set token contract?"
243 : "Set payout address?",
244 COL_TEXT, &font_pjs_20_medium);
245 lv_obj_t *l = ota_text(card, t, label, COL_DIM, &font_inter_14);
246 lv_obj_t *a = ota_text(card, l, addr, COL_TEXT, &font_inter_14_medium);
247 /* Kept to the sentence that changes a decision: the card grows only up to
248 * ota_fit_card's ceiling of one screen, past which the buttons overlap the
249 * text. */
250 lv_obj_t *warn = ota_text(card, a,
251 contract
252 ? "A wrong contract charges a different asset."
253 : "Takings will be sent here. Check it against "
254 "your own records.",
255 COL_DIM, &font_inter_14);
256 ota_fit_card(card, warn, 86); /* two 40px buttons at -46 and -2 */
257
258 /* Reject is the wide, plainly-labelled one and Accept is the deliberate tap:
259 * the safe answer to "a stranger's address appeared on my terminal" is no. */
260 (void)make_button(card, "Accept", COL_ACCENT, COL_BG, OTA_TEXT_W, 40,
261 LV_ALIGN_BOTTOM_MID, 0, -46, ACT_PROV_OK,
262 &font_pjs_20_semibold);
263 (void)make_button(card, "Reject", COL_SURFACE, COL_TEXT, OTA_TEXT_W, 40,
