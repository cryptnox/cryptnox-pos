145static inline const pos_asset_t *pos_asset_of(pos_chain_t chain)
146{
147 for (size_t i = 0U; i < POS_ASSET_COUNT; i++) {
148 if (POS_ASSETS[i].chain == chain) { return &POS_ASSETS[i]; }
149 }
150 return &POS_ASSETS[0];
151}

183static inline bool pos_chain_is_native_evm(pos_chain_t c)

190static inline const pos_net_info_t *pos_net_info(pos_net_t net)
