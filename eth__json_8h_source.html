135static inline bool eth_json_hex_quantity(const char *hex, uint64_t *out)
136{
137 if ((hex == NULL) || (out == NULL)) { return false; }
138 if ((hex[0] != '0') || ((hex[1] != 'x') && (hex[1] != 'X'))) { return false; }
139
140 const char *p = &hex[2];
141 if (*p == '\0') { return false; } /* "0x" on its own is not a quantity */
142
143 /* Every digit is checked, leading zeros included: validating only the
144 * significant ones would read "0x00zz" as zero rather than as garbage. */
145 size_t len = 0U;
146 while (p[len] != '\0') {
147 const char c = p[len];
148 const bool is_hex = ((c >= '0') && (c <= '9')) ||
149 ((c >= 'a') && (c <= 'f')) ||
150 ((c >= 'A') && (c <= 'F'));
151 if (!is_hex) { return false; }
152 len++;
153 }
154
155 /* Nodes pad a uint256 out to 64 characters; strip down to the significant
156 * digits so the width test below is about the value, not the encoding.
157 * One digit always survives, so "0x000" stays parseable as zero. */
158 while ((*p == '0') && (p[1] != '\0')) { p++; len--; }
159
160 if (len > 16U) { *out = UINT64_MAX; return true; }
161
162 uint64_t v = 0U;
163 for (size_t i = 0U; i < len; i++) {
164 const char c = p[i];
165 uint8_t n;
166 if ((c >= '0') && (c <= '9')) { n = (uint8_t)(c - '0'); }
167 else if ((c >= 'a') && (c <= 'f')) { n = (uint8_t)((c - 'a') + 10); }
168 else { n = (uint8_t)((c - 'A') + 10); }
169 v = (v << 4) | (uint64_t)n;
170 }
171 *out = v;
172 return true;
173}
