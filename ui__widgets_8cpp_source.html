39 * (Compare the raw RGB565 value — lv_color_eq isn't in this LVGL build.) */

59 lv_obj_set_style_text_color(lbl, fg, LV_PART_MAIN);

121/* Coin mark, optionally with a network chip on its bottom-right corner, the way
122 * wallets badge a token with the chain it lives on — so "USDC" (Sepolia) and a

124 * carry their white ring in the bitmap (tools/gen_chain_icons.py).

126 * @p chip may be NULL, and the box then shrinks to COIN_SZ. The coin is centred

414/* The UTC offset the clock adds, cached: settings_get_tz_offset_min() opens
415 * NVS and clock_refresh() runs every three seconds. Reloaded through
416 * ui_clock_changed() when the config page stores a new one. */
417static int16_t s_tz_off_min = 0;
418static uint8_t s_tz_dst = 0;
419volatile bool s_tz_dirty = true;
420
431static void clock_refresh(void) {
432 if (s_clock_lbl == NULL) { return; }
433
434 if (s_tz_dirty) {
435 s_tz_dirty = false;
436 s_tz_off_min = settings_get_tz_offset_min();
437 s_tz_dst = settings_get_tz_dst();
438 }
439
440 const time_t utc = time(NULL);
441 /* Nov 2023, before any real sale. Tested on the unshifted instant, so a
442 * negative offset cannot drag a just-set clock back under the threshold. */
443 if (utc < (time_t)1700000000) {
444 lv_label_set_text(s_clock_lbl, "--:--");
445 return;
446 }
447
448 const time_t local = utc + ((time_t)civil_local_offset_min(
449 utc, s_tz_off_min, s_tz_dst) * 60);
450 struct tm tmv;
451 if (gmtime_r(&local, &tmv) == NULL) {
452 lv_label_set_text(s_clock_lbl, "--:--");
453 return;

622/******************************************************************
623 * 8b. Wi-Fi signal — the wave, top right of every screen
624 *
625 * Three arcs over a dot, read at a glance from across a counter. The dot is at
626 * least 4px: LVGL clamps the radius to half the side, so 3px draws a square.
627 *
628 * The sizes are one system. The two gaps must be equal, or the dot reads as
629 * stuck to the first ring:
630 * dot to first ring: SIG_R0 - (SIG_AW / 2) - (SIG_DOT / 2)
631 * ring to ring: SIG_RSTEP - SIG_AW
632 *
633 * On the top layer so it survives screen swaps; a modal, created on the same
634 * layer later, covers it.
635 ******************************************************************/
636#define SIG_ARCS 3
637#define SIG_AW 2 /* arc thickness */
638/* Mark is 22x13; both gaps above are 1px. At r=4 the inner ring's 140-degree
639 * cut still bends 2.6px, so it reads as an arc rather than as a dash. */
640#define SIG_R0 4 /* innermost arc radius; each ring adds SIG_RSTEP */
641#define SIG_RSTEP 3
642/* Must stay even: an arc object is 2r + SIG_AW wide, so the fan's centre is a
643 * whole coordinate, and only an even dot centres on it (odd sits half a pixel
644 * off). */
645#define SIG_DOT 4
646/* Per-ring sweep, innermost first, centred on 12 o'clock. Inner rings get more
647 * of their circle so their sagitta r·(1 - cos(sweep/2)) reads as a curve, not a
648 * dash. The sweeps also set the icon's width (2r·sin(sweep/2)) without growing
649 * its height. Past ~140° the inner ring looks like most of a circle; to widen
650 * further, raise SIG_R0. */
651static const uint16_t SIG_SWEEP[SIG_ARCS] = { 140, 110, 100 };
652
653/* The outermost ring's outer edge, which is the icon's half-width and, with the
654 * dot's bottom half, its height. Everything below is derived from it. */
655#define SIG_REACH (SIG_R0 + ((SIG_ARCS - 1) * SIG_RSTEP) + (SIG_AW / 2))
656#define SIG_W (2 * SIG_REACH)
657#define SIG_H (SIG_REACH + ((SIG_DOT + 1) / 2))
658/* Inset from the right edge. */
659#define SIG_INSET 14
660
661static lv_obj_t *s_sig_arc[SIG_ARCS];
662static lv_obj_t *s_sig_dot = NULL;
663static lv_obj_t *s_sig_box = NULL;
664
665static void signal_refresh(lv_timer_t *t) {
666 (void)t;
667 int8_t rssi = 0;
668 /* net_wifi_rssi() fails when the station is not associated — grey dot, no
669 * arc, which is the answer the operator needs before blaming the card.
670 * The cuts match the words on the settings page (Good / Fair / Weak), so
671 * the icon and the "Signal" field there cannot disagree. */
672 const bool up = net_wifi_rssi(&rssi);
673 const int lit = !up ? 0 : (rssi >= -60) ? 3 : (rssi >= -70) ? 2 : 1;
674 for (int i = 0; i < SIG_ARCS; i++) {
675 /* Unlit is COL_DIM, not COL_BORDER: the sale flow's page IS COL_BORDER,
676 * so a COL_BORDER arc would vanish into it. See COL_PAGE. */
677 lv_obj_set_style_arc_color(s_sig_arc[i],
678 (i < lit) ? COL_TEXT : COL_DIM,
679 LV_PART_MAIN);
680 }
681 /* Amber while the network is down and being re-joined in the background
682 * (net.cpp): it tells the operator to wait rather than to go and fix
683 * something. */
684 lv_obj_set_style_bg_color(s_sig_dot,
685 up ? COL_TEXT
686 : net_wifi_reconnecting() ? COL_WARN : COL_DIM,
687 LV_PART_MAIN);
688 /* Hidden on the splash (nothing connected yet), the calibration screen (it
689 * would cover a corner target), and the admin screens, whose tab bar owns
690 * the top 42px; the Wi-Fi tab reports the signal in words instead. */
691 const bool show = (s_req_screen != UI_SCREEN_SPLASH) &&
692 (s_req_screen != UI_SCREEN_TOUCH_CAL) &&
693 (s_req_screen != UI_SCREEN_SETTINGS) &&
694 (s_req_screen != UI_SCREEN_ADMIN_UNLOCK) &&
695 (s_req_screen != UI_SCREEN_ADMIN_SET) &&
696 /* the first-run setup flow: nothing to report yet */
697 (s_req_screen != UI_SCREEN_WELCOME) &&
698 (s_req_screen != UI_SCREEN_PROV) &&
699 (s_req_screen != UI_SCREEN_WIFI_CONNECTING) &&
700 (s_req_screen != UI_SCREEN_WIFI_LIST) &&
701 (s_req_screen != UI_SCREEN_WIFI_PASS) &&
702 (s_req_screen != UI_SCREEN_CARD_WAIT) &&
703 !((s_req_screen == UI_SCREEN_PIN) && s_pin_for_card);
704 if (show) { lv_obj_clear_flag(s_sig_box, LV_OBJ_FLAG_HIDDEN); }
705 else { lv_obj_add_flag(s_sig_box, LV_OBJ_FLAG_HIDDEN); }
706
707 /* Same cadence, same reason — the band's two occupants refresh together. */
708 clock_refresh();
709}
710void signal_init(void) {
711 s_sig_box = lv_obj_create(lv_layer_top());
712 lv_obj_remove_style_all(s_sig_box);
713 lv_obj_set_size(s_sig_box, SIG_W, SIG_H);
714 lv_obj_align(s_sig_box, LV_ALIGN_TOP_RIGHT, -SIG_INSET, 6);
715 lv_obj_clear_flag(s_sig_box, LV_OBJ_FLAG_SCROLLABLE);
716 lv_obj_clear_flag(s_sig_box, LV_OBJ_FLAG_CLICKABLE);
717
718 /* Every ring is struck from the same point — the dot — so each arc object is
719 * sized to its own ring and then centred on that point, not on the box. */
720 const lv_coord_t cx = SIG_W / 2, cy = SIG_REACH;
721 for (int i = 0; i < SIG_ARCS; i++) {
722 const lv_coord_t d = (2 * (SIG_R0 + (i * SIG_RSTEP))) + SIG_AW;
723 lv_obj_t *a = lv_arc_create(s_sig_box);
724 lv_obj_remove_style_all(a); /* also kills the knob and the
725 * value indicator: both draw at
726 * the default arc width of 0 */
727 lv_obj_set_size(a, d, d);
728 lv_obj_set_pos(a, cx - (d / 2), cy - (d / 2));
729 /* 0° is 3 o'clock and angles run clockwise, so 270 is 12 o'clock and each
730 * ring is cut symmetrically about it. */
731 lv_arc_set_bg_angles(a, 270 - (SIG_SWEEP[i] / 2), 270 + (SIG_SWEEP[i] / 2));
732 lv_obj_set_style_arc_width(a, SIG_AW, LV_PART_MAIN);
733 /* Square-cut: a rounded cap on a 2px stroke anti-aliases into a darker
734 * pixel past each tip. */
735 lv_obj_set_style_arc_rounded(a, false, LV_PART_MAIN);
736 lv_obj_clear_flag(a, LV_OBJ_FLAG_CLICKABLE);
737 s_sig_arc[i] = a;
738 }
739
740 s_sig_dot = lv_obj_create(s_sig_box);
741 lv_obj_remove_style_all(s_sig_dot);
742 lv_obj_set_size(s_sig_dot, SIG_DOT, SIG_DOT);
743 lv_obj_set_pos(s_sig_dot, cx - (SIG_DOT / 2), cy - (SIG_DOT / 2));
744 lv_obj_set_style_bg_opa(s_sig_dot, LV_OPA_COVER, LV_PART_MAIN);
745 lv_obj_set_style_radius(s_sig_dot, LV_RADIUS_CIRCLE, LV_PART_MAIN);
746 lv_obj_clear_flag(s_sig_dot, LV_OBJ_FLAG_CLICKABLE);
747
748 /* ponytail: 3 s poll. An event-driven update would mean a Wi-Fi handler
749 * poking LVGL from the event task — add one only if the lag shows. */
750 (void)lv_timer_create(signal_refresh, 3000, NULL);
751 signal_refresh(NULL);
752}
