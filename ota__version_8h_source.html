48static inline void ota_version_parse(const char *s,
49 unsigned long out[OTA_VERSION_PARTS])
50{
51 for (size_t i = 0U; i < OTA_VERSION_PARTS; i++) { out[i] = 0UL; }
52 if (s == NULL) { return; }
53
54 size_t i = 0U;
55 if ((s[0] == 'v') || (s[0] == 'V')) { i = 1U; }
56
57 size_t part = 0U;
58 bool any = false; /* digits seen in the current component */
59 for (; (i < OTA_VERSION_MAX) && (s[i] != '\0'); i++) {
60 const char c = s[i];
61 if ((c >= '0') && (c <= '9')) {
62 /* Clamp rather than wrap: a component this large is junk, and
63 * "compares as very large" beats "compares as whatever the
64 * overflow happened to leave behind". */
65 if (out[part] < 100000000UL) {
66 out[part] = (out[part] * 10UL) + (unsigned long)(c - '0');
67 }
68 any = true;
69 } else if (c == '.') {
70 /* A dot with no digits before it ends the number entirely — ".."
71 * and a leading dot are malformed, not zero-valued components. */
72 if (!any) { break; }
73 part++;
74 any = false;
75 if (part >= OTA_VERSION_PARTS) { break; }
76 } else {
77 break; /* pre-release or build suffix: ignored */
78 }
79 }
80}

90static inline int ota_version_cmp(const char *a, const char *b)
91{
92 unsigned long va[OTA_VERSION_PARTS];
93 unsigned long vb[OTA_VERSION_PARTS];
94 ota_version_parse(a, va);
95 ota_version_parse(b, vb);
96
97 for (size_t i = 0U; i < OTA_VERSION_PARTS; i++) {
98 if (va[i] != vb[i]) { return (va[i] < vb[i]) ? -1 : 1; }
99 }
100 return 0;
101}
