70static inline card_state_t card_state(const uint8_t *r, size_t n)
71{
72 if ((r == NULL) || (n < CARD_SELECT_MIN)) { return CARD_UNKNOWN; }
73 /* Not 0x9000 means the applet did not answer this SELECT, so there is no
74 * layout below to trust. */
75 if ((r[n - 2U] != 0x90U) || (r[n - 1U] != 0x00U)) { return CARD_UNKNOWN; }
76 /* A card type whose flags byte we have not been told about could have
77 * anything in bit 6, and reading it wrong here would refuse a card that
78 * works. 'B' is Basic, 'N' is NFT; both carry these two flags. */
79 if ((r[0] != (uint8_t)'B') && (r[0] != (uint8_t)'N')) { return CARD_UNKNOWN; }
80
81 const uint8_t flags = r[CARD_FLAGS_OFF];
82 if ((flags & CARD_FLAG_PIN_SET) == 0U) { return CARD_NO_PIN; }
83 if ((flags & CARD_FLAG_SEEDED) == 0U) { return CARD_NO_KEY; }
84 return CARD_READY;
85}
