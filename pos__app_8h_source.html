144/* What a broadcast attempt ended as. UNKNOWN: the node never answered, so the
145 * transaction may be on its way. Only the chain can settle it, and until it does
146 * the sale is Unconfirmed — never declined, because a declined sale is one the
147 * merchant takes again. */

149 BCAST_FAILED = 0,
150 BCAST_SENT,
151 BCAST_UNKNOWN,
152} bcast_t;

154/* Everything the receipt poll needs, kept here so "Check again" on the

156 * (inflight_persist), so its layout is the record's. */

170} inflight_t;
